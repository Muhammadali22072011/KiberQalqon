① 08_samples.sql — приватный бакет для образцов malware 
-- KiberQalqon — 08_samples.sql
-- Xavfli/shubhali APK NAMUNALARINI serverda yig'ish (panelda yuklab olib o'rganish uchun).
-- Idempotent: bir necha marta ishga tushirsa ham xavfsiz.

insert into storage.buckets (id, name, public, file_size_limit, allowed_mime_types)
values (
  'malware-samples',
  'malware-samples',
  false,
  52428800,  -- 50 MB
  array[
    'application/vnd.android.package-archive',
    'application/octet-stream',
    'application/zip'
  ]
)
on conflict (id) do update set
  public             = false,
  file_size_limit    = excluded.file_size_limit,
  allowed_mime_types = excluded.allowed_mime_types;
② 09_totp_replay.sql — защита 2FA-кода от повторного использования 
-- KiberQalqon — 09_totp_replay.sql  (TOTP replay himoyasi)
-- 2FA kodi bir martagina ishlatilsin (o'g'irlangan kod qayta kira olmasin).
-- Idempotent: bir necha marta ishga tushirsa ham xavfsiz.

create table if not exists auth_totp (
  id            text primary key,        -- bitta egasi: 'owner'
  last_counter  bigint not null default 0,
  used_at       timestamptz not null default now()
);

-- RLS: anon hech narsa ko'rmaydi, server service_role bilan ishlaydi.
alter table auth_totp enable row level security;
