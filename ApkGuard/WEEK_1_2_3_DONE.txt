### 1. Запустить WorkManager в MainActivity:

Добавить в onCreate():
```kotlin
override fun onCreate(savedInstanceState: Bundle?) {
    super.onCreate(savedInstanceState)
    
    // ... существующий код ...
    
    // Запускаем периодическую проверку
    startPeriodicCheck()
}

private fun startPeriodicCheck() {
    val constraints = Constraints.Builder()
        .setRequiresBatteryNotLow(true) // Только если батарея не низкая
        .build()
    
    val periodicWork = PeriodicWorkRequestBuilder<PeriodicCheckWorker>(
        15, TimeUnit.MINUTES // Каждые 15 минут
    )
        .setConstraints(constraints)
        .build()
    
    WorkManager.getInstance(this).enqueueUniquePeriodicWork(
        "periodic_apk_check",
        ExistingPeriodicWorkPolicy.KEEP,
        periodicWork
    )
    
    Log.d("MainActivity", "✅ Периодическая проверка запущена")
}
```

### 2. Диалог первого запуска:

Создать FirstRunDialog.kt:
```kotlin
class FirstRunDialog(context: Context) : Dialog(context) {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.dialog_first_run)
        
        findViewById<Button>(R.id.btnGoSettings).setOnClickListener {
            // Открыть настройки
            context.startActivity(Intent(context, SettingsActivity::class.java))
            Config.setFirstRunComplete(context)
            dismiss()
        }
        
        findViewById<Button>(R.id.btnLater).setOnClickListener {
            Config.setFirstRunComplete(context)
            dismiss()
        }
    }
}
```

Показать в MainActivity:
```kotlin
if (Config.isFirstRun(this)) {
    FirstRunDialog(this).show()
}
```
